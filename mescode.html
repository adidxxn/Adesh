#SingleInstance Force
#NoEnv
#Persistent

SendMode Input
SetBatchLines, -1
SetKeyDelay, 0
SetWinDelay, 0
SetControlDelay, 0
SetMouseDelay, 0
SetTitleMatchMode, 2

paused := false
count := 0
startRow := 0

;====================================
; F8 = Excel Selection Copy + Save Row
;====================================
F8::
try
{
    xl := ComObjActive("Excel.Application")

    startRow := xl.Selection.Row

    Clipboard := ""
    xl.Selection.Copy()

    ClipWait, 1

    ToolTip, Data Copied Successfully`nStart Row: %startRow%
    SetTimer, RemoveToolTip, -1500
}
catch
{
    MsgBox, 48, Error, Excel open nahi hai ya data select nahi hai.
}
Return

;====================================
; F9 = Start Processing
;====================================
F9::

if !WinActive("RM Quality")
{
    MsgBox, 48, Warning, RM Quality Window Active Karo.
    Return
}

if (startRow = 0)
{
    MsgBox, 48, Warning, Pehle Excel me data select karke F8 dabao.
    Return
}

data := Clipboard

if (data = "")
{
    MsgBox, 48, Error, Clipboard Empty Hai.
    Return
}

try
{
    xl := ComObjActive("Excel.Application")
}
catch
{
    MsgBox, 48, Error, Excel Open Nahi Hai.
    Return
}

rows := StrSplit(data, "`n")
count := 0

for index, row in rows
{
    while (paused)
        Sleep, 100

    line := Trim(row, "`r`n`t ")

    if (line = "")
        Continue

    cols := StrSplit(line, A_Tab)

    if (cols.MaxIndex() >= 1 && Trim(cols[1]) != "")
{
    SendInput % cols[1]
    Sleep, 80
    SendInput {Enter}

    Sleep, 500

ControlGetText, errorA, WindowsForms10.STATIC.app.0.13965fa_r7_ad15, RM Quality

if (Trim(errorA) != "")
{
    cellD := xl.ActiveSheet.Cells(startRow + count, 4)

    cellD.Value := errorA
    cellD.WrapText := False
    cellD.Font.Bold := True

    if (InStr(errorA, "Quality") || InStr(errorA, "Already"))
    {
        ; Green Fill
        cellD.Interior.Color := 13434828

        ; Dark Green Text
        cellD.Font.Color := 32768
    }
    else
    {
        ; Red Fill
        cellD.Interior.Color := 13551615

        ; Dark Red Text
        cellD.Font.Color := 192
    }
}
    Sleep, 60
}

if (cols.MaxIndex() >= 2 && Trim(cols[2]) != "")
{
    SendInput % cols[2]
    Sleep, 60

    SendInput {Enter}
    Sleep, 60

    SendInput {Enter}
    Sleep, 20

    SendInput {Enter}
    Sleep, 30

    SendInput {Enter}

    Sleep, 500

ControlGetText, errorB, WindowsForms10.STATIC.app.0.13965fa_r7_ad15, RM Quality

if (Trim(errorB) != "")
{
    cellE := xl.ActiveSheet.Cells(startRow + count, 5)

    cellE.Value := errorB
    cellE.WrapText := False
    cellE.Font.Bold := True

    if (InStr(errorB, "Quality") || InStr(errorB, "Already"))
    {
        ; Green Fill
        cellE.Interior.Color := 13434828

        ; Dark Green Text
        cellE.Font.Color := 32768
    }
    else
    {
        ; Red Fill
        cellE.Interior.Color := 13551615

        ; Dark Red Text
        cellE.Font.Color := 192
    }
}
    Sleep, 300
}

    ;=====================
    ; Row Complete
    ;=====================
    count++

    rowNum := startRow + count - 1

try
{
    FormatTime, currentTime,, dd-MM-yyyy HH:mm:ss

    statusCell := xl.ActiveSheet.Cells(rowNum, 3)

    statusCell.Value := "ROW PROCESSED by @Adesh - " . currentTime

    ; Green Fill
    statusCell.Interior.Color := 13434828

    ; Dark Green Text
    statusCell.Font.Color := 32768
    statusCell.Font.Bold := True

    ;-------------------------
    ; D & E Column Auto Width
    ;-------------------------
    xl.ActiveSheet.Columns("C").AutoFit
    xl.ActiveSheet.Columns("D").AutoFit
    xl.ActiveSheet.Columns("E").AutoFit

    
    ; Auto Row Height
    xl.ActiveSheet.Rows(rowNum).AutoFit

    ; Scroll Down
    visibleRows := xl.ActiveWindow.VisibleRange.Rows.Count

    if (rowNum >= xl.ActiveWindow.ScrollRow + visibleRows - 3)
        xl.ActiveWindow.ScrollRow := rowNum - visibleRows + 3
}
catch
{
}    ToolTip, Completed Rows: %count%
}

ToolTip

MsgBox, 64, @Adesh, Total Rows Processed: %count%

Return

;====================================
; F10 = Pause / Resume
;====================================
F10::

paused := !paused

if (paused)
    ToolTip, PAUSED
else
    ToolTip, RESUMED

SetTimer, RemoveToolTip, -1000

Return

;====================================
RemoveToolTip:
ToolTip
Return
