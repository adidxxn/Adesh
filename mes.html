SetControlDelay, -1
SetMouseDelay, -1
SetTitleMatchMode, 2
Process, Priority,, High
ListLines Off

paused := false
count := 0
success := 0
failed := 0
startRow := 0
totalRows := 0
startTime := 0

F9::

if !WinExist("RM Quality")
{
    MsgBox, 48, Warning, RM Quality Window Open Karo.
    Return
}

try
{
    xl := ComObjActive("Excel.Application")
}
catch
{
    MsgBox, 48, Error, Excel Open Nahi Hai.
    Return
}

sel := xl.Selection

if !IsObject(sel)
{
    MsgBox, 48, Error, Excel Selection Nahi Mili.
    Return
}

startRow  := sel.Row
totalRows := sel.Rows.Count
startRow  := sel.Row
totalRows := sel.Rows.Count
try
{
    xl.ScreenUpdating := False

    sht := xl.ActiveSheet
    sht.Columns("A:E").Font.Size := 8

    xl.ScreenUpdating := True
}
catch
{
}

count := 0
success := 0
failed := 0

startTime := A_TickCount

Loop % totalRows
{
    while (paused)
        Sleep,100
barcodePass := 0
mpnPass := 0

    excelRow := startRow + A_Index - 1

    col1 := Trim(xl.ActiveSheet.Cells(excelRow,1).Text)
    col2 := Trim(xl.ActiveSheet.Cells(excelRow,2).Text)

    ; Blank Row Skip
    if (col1="" && col2="")
        Continue

    count++

    ToolTip,
(
 
   @Adesh  %count% / %totalRows%

Dixon Barcode :
%col1%
MPN :
%col2%
)

if (col1 != "")
{
    SendInput {Raw}%col1%
    Sleep,20

    SendInput {Enter}

    Sleep,120

    ControlGetText,errorA,WindowsForms10.STATIC.app.0.13965fa_r7_ad15,RM Quality

    if (Trim(errorA)="")
    {
        barcodePass := 1
    }
    else
    {
        barcodePass := 0

        cellD := xl.ActiveSheet.Cells(excelRow,4)

        cellD.Value := errorA
        cellD.WrapText := False
        cellD.Font.Bold := True

        if (InStr(errorA,"Quality") || InStr(errorA,"Already"))
        {
            cellD.Interior.Color := 13434828
            cellD.Font.Color := 32768
        }
        else
        {
            cellD.Interior.Color := 13551615
            cellD.Font.Color := 192
        }
    }
}

if (col2 != "")
{
    SendInput %col2%

    Sleep,20

    SendInput {Enter}
    Sleep,10

    SendInput {Enter}
    Sleep,10

    SendInput {Enter}
    Sleep,10

    SendInput {Enter}

    Sleep,120

    ControlGetText,errorB,WindowsForms10.STATIC.app.0.13965fa_r7_ad15,RM Quality

    if (Trim(errorB)="")
    {
        mpnPass := 1
    }
    else
    {
        mpnPass := 0

        cellE := xl.ActiveSheet.Cells(excelRow,5)

        cellE.Value := errorB
        cellE.WrapText := False
        cellE.Font.Bold := True

        if (InStr(errorB,"Quality") || InStr(errorB,"Already"))
        {
            cellE.Interior.Color := 13434828
            cellE.Font.Color := 32768
        }
        else
        {
            cellE.Interior.Color := 13551615
            cellE.Font.Color := 192
        }
    }
}

FormatTime,currentTime,,dd-MM-yyyy HH:mm:ss

cellC := xl.ActiveSheet.Cells(excelRow,3)

cellC.Value := "ROW PROCESSED by @Adesh - " . currentTime

cellC.Font.Bold := True
cellC.Font.Color := 32768
cellC.Interior.Color := 13434828

try
{
    sht := xl.ActiveSheet

    sht.Columns("C").AutoFit
    sht.Columns("D").ColumnWidth := 35    
    sht.Columns("E").AutoFit

    sht.Rows(excelRow).AutoFit
}
catch
{
}

try
{
    visibleRows := xl.ActiveWindow.VisibleRange.Rows.Count

    if (excelRow >= xl.ActiveWindow.ScrollRow + visibleRows - 3)
        xl.ActiveWindow.ScrollRow := excelRow - visibleRows + 3
}
catch
{
}

if (barcodePass && mpnPass)
    success++
else
    failed++

if (Mod(count,10)=0)
{
    try
        xl.ActiveWorkbook.Save()
    catch
    {
    }
}

}

try
{
    xl.ActiveWorkbook.Save()
}
catch
{
}

try
{
    xl.ScreenUpdating := True

    sht := xl.ActiveSheet

    sht.Columns("C").EntireColumn.AutoFit
    sht.Columns("D").ColumnWidth := 35
    sht.Columns("E").EntireColumn.AutoFit

    sht.Rows.EntireRow.AutoFit

    xl.ScreenUpdating := True
}
catch
{
}

FormatTime, d,, dd-MMM-yyyy HH-mm-ss


try
{
    xl.ScreenUpdating := False

    xl.ActiveSheet.Name := "MES PASS " . d

    xl.ActiveWorkbook.Save()

    xl.ScreenUpdating := True
}
catch e
{
    xl.ScreenUpdating := True
    MsgBox, 48, Rename Error, % e.Message
}

ToolTip

elapsed := A_TickCount-startTime

totalSec := Floor(elapsed/1000)

hh := Floor(totalSec/3600)
mm := Floor(Mod(totalSec,3600)/60)
ss := Mod(totalSec,60)

timeTaken := Format("{:02}:{:02}:{:02}",hh,mm,ss)

MsgBox,64,@Adesh,
(
-----------------------------------
RM QUALITY PROCESS COMPLETED
-----------------------------------
Total Rows  : %totalRows%
Time        : %timeTaken%
-----------------------------------
)

Return
